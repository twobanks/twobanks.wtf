import polyline from "@mapbox/polyline"
import fs from "fs/promises"
import path from "path"
export interface ParsedGpxTrack {
coordinates: [number, number][] // [lng, lat] para Mapbox
encodedPolyline: string // [lat, lng] codificado para StravaMap
elevationPoints: {
distanceKm: number
elevationM: number
}[]
totalDistanceMeters: number
totalElevationGain: number
}
function haversineMeters(
lat1: number,
lon1: number,
lat2: number,
lon2: number
): number {
const R = 6371000
const toRad = (deg: number) => (deg * Math.PI) / 180
const dLat = toRad(lat2 - lat1)
const dLon = toRad(lon2 - lon1)
const a =
Math.sin(dLat / 2) * Math.sin(dLat / 2) +
Math.cos(toRad(lat1)) *
Math.cos(toRad(lat2)) *
Math.sin(dLon / 2) *
Math.sin(dLon / 2)
return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a))
}
function sampleArray<T>(items: T[], maxPoints: number): T[] {
if (items.length <= maxPoints) return items
const step = (items.length - 1) / (maxPoints - 1)
const sampled: T[] = []
for (let i = 0; i < maxPoints; i++) {
sampled.push(items[Math.round(i * step)])
}
return sampled
}
export async function parsePublicGpxFile(
publicRelativePath?: string,
maxPoints = 500
): Promise<ParsedGpxTrack | null> {
if (!publicRelativePath) return null
try {
const cleanPath = publicRelativePath.replace(/^\/+/, "")
const fullPath = path.join(process.cwd(), "public", cleanPath)
const xml = await fs.readFile(fullPath, "utf-8")
const pointRegex = /<(?:trkpt|rtept)\b([^>]*)>([\s\S]*?)<\/(?:trkpt|rtept)>/gi
const rawPoints: { lat: number; lon: number; ele: number | null }[] = []
let match: RegExpExecArray | null
while ((match = pointRegex.exec(xml)) !== null) {
const attrs = match[1]
const body = match[2]
const latMatch = /\blat=["']([^"']+)["']/i.exec(attrs)
const lonMatch = /\blon=["']([^"']+)["']/i.exec(attrs)
if (!latMatch || !lonMatch) continue
const lat = Number.parseFloat(latMatch[1])
const lon = Number.parseFloat(lonMatch[1])
if (Number.isNaN(lat) || Number.isNaN(lon)) continue
const eleMatch = /<ele>([^<]+)<\/ele>/i.exec(body)
const ele = eleMatch ? Number.parseFloat(eleMatch[1]) : null
rawPoints.push({
lat,
lon,
ele: ele !== null && !Number.isNaN(ele) ? ele : null,
})
}
if (rawPoints.length === 0) return null
let cumulativeMeters = 0
let totalGain = 0
const enrichedPoints: {
lat: number
lon: number
ele: number | null
distMeters: number
}[] = []
for (let i = 0; i < rawPoints.length; i++) {
const curr = rawPoints[i]
if (i > 0) {
const prev = rawPoints[i - 1]
cumulativeMeters += haversineMeters(prev.lat, prev.lon, curr.lat, curr.lon)
if (curr.ele !== null && prev.ele !== null) {
const diff = curr.ele - prev.ele
if (diff > 0.5) totalGain += diff // filtro leve de ruído barométrico
}
}
enrichedPoints.push({ ...curr, distMeters: cumulativeMeters })
}
const sampled = sampleArray(enrichedPoints, maxPoints)
const coordinates: [number, number][] = sampled.map((p) => [p.lon, p.lat])
const encodedPolyline = polyline.encode(sampled.map((p) => [p.lat, p.lon]))
const elevationPoints = sampled
.filter((p): p is typeof p & { ele: number } => p.ele !== null)
.map((p) => ({
distanceKm: Number((p.distMeters / 1000).toFixed(1)),
elevationM: Math.round(p.ele),
}))
return {
coordinates,
encodedPolyline,
elevationPoints,
totalDistanceMeters: Math.round(cumulativeMeters),
totalElevationGain: Math.round(totalGain),
}
} catch {
return null
}
}

}) {

const endDate = Math.floor(Date.now() / 1000)
let startDate = 1356998400 // 2013-01-01 (para fullSync)

// Busca a data do treino mais recente no banco (ex.: agosto) como ponto de partida
const [latest] = await db
.select({ startDate: stravaWorkouts.startDate })
.from(stravaWorkouts)
.orderBy(desc(stravaWorkouts.startDate))
.limit(1)

startDate = Math.max(0, Math.floor(parsedMs / 1000) - 3 * 24 * 60 * 60)
}
}
}
let totalSincronizados = 0
let page = 1
while (true) {
const activitiesResponse = await fetch(
`https://www.strava.com/api/v3/athlete/activities?after=${startDate}&before=${endDate}&page=${page}&per_page=100`,
{
headers: { Authorization: `Bearer ${token}` },
cache: "no-store",
}
)
if (!activitiesResponse.ok) {
const errText = await activitiesResponse.text()
throw new Error(`Erro ao listar atividades (${activitiesResponse.status}): ${errText}`)
}
const activities = await activitiesResponse.json()
if (!Array.isArray(activities) || activities.length === 0) break
for (const activity of activities) {
await upsertActivityRecord(activity, token, {
userId: options?.userId,
skipStreamsIfExists: true,
})
totalSincronizados++
await delay(150)
}
if (activities.length < 100) break
page++
}
await fetchAndSaveAllGearDetails()
return {
totalActivities: totalSincronizados,
startDate: new Date(startDate * 1000).toISOString(),
endDate: new Date(endDate * 1000).toISOString(),
}
}

const seconds = Number(row.totalMovingTime) || 0
const count = Number(row.count) || 0
if (RUN_TYPES.includes(workoutType)) {
runMeters += dist
runAndRideSeconds += seconds
} else if (RIDE_TYPES.includes(workoutType)) {
rideMeters += dist
runAndRideSeconds += seconds
} else {
otherMeters += dist
}
activitiesCount += count
}
const runKm = Math.round(runMeters / 1000)
const rideKm = Math.round(rideMeters / 1000)
const totalKm = Math.round((runMeters + rideMeters + otherMeters) / 1000)
const totalHours = Number((runAndRideSeconds / 3600).toFixed(1))
return {
totalKm,
runKm,
rideKm,
totalHours,
activitiesCount,
}
}

async function hydrateTrip(
trip: Trip,
workoutsMap: Map<string, SanitizedWorkout>,
bikeGear: EnrichedTrip["bikeGear"] = null
): Promise<EnrichedTrip> {
const parsedDayGpxs = await Promise.all(
trip.days.map((day) => parsePublicGpxFile(day.plannedGpxFile, 450))
)
const plannedSegments: [number, number][][] = []
const computedElevationProfile: {
distanceKm: number
elevationM: number
dayLabel?: string
}[] = []
let accumulatedKmOffset = 0
const enrichedDays: EnrichedTripDay[] = trip.days.map((day, idx) => {
const gpx = parsedDayGpxs[idx]
if (gpx && gpx.coordinates.length > 0) {
plannedSegments.push(gpx.coordinates)
if (gpx.elevationPoints.length > 0) {
gpx.elevationPoints.forEach((pt, ptIdx) => {
computedElevationProfile.push({
distanceKm: Number((accumulatedKmOffset + pt.distanceKm).toFixed(1)),
elevationM: pt.elevationM,
dayLabel: ptIdx === 0 ? `Dia ${day.dayNumber}` : undefined,
})
})
}
accumulatedKmOffset +=
(day.plannedDistanceMeters || gpx.totalDistanceMeters) / 1000
}
const dayWorkouts = day.stravaWorkoutIds
.map((id) => workoutsMap.get(id))
.filter((w): w is SanitizedWorkout => Boolean(w))
const actualDistanceMeters = dayWorkouts.reduce(
(acc, w) => acc + w.distanceMeters,
0
)
const actualElevationGain = dayWorkouts.reduce(
(acc, w) => acc + w.elevationGainMeters,
0
)
const actualMovingTimeSeconds = dayWorkouts.reduce(
(acc, w) => acc + w.movingTimeSeconds,
0
)
const actualAverageSpeedMs =
actualMovingTimeSeconds > 0
? actualDistanceMeters / actualMovingTimeSeconds
: 0
const isCompleted =
day.dayType === "ride"
? dayWorkouts.length > 0 || Boolean(day.completedManual)
: Boolean(day.completedManual)
return {
...day,
plannedPolyline: day.plannedPolyline || gpx?.encodedPolyline,
workouts: dayWorkouts,
isCompleted,
actualDistanceMeters,
actualElevationGain,
actualMovingTimeSeconds,
actualAverageSpeedMs,
}
})
const actualTotalDistanceMeters = enrichedDays.reduce(
(acc, d) => acc + d.actualDistanceMeters,
0
)
const actualTotalElevationGain = enrichedDays.reduce(
(acc, d) => acc + d.actualElevationGain,
0
)
const actualTotalMovingTimeSeconds = enrichedDays.reduce(
(acc, d) => acc + d.actualMovingTimeSeconds,
0
)
const rideDays = enrichedDays.filter((d) => d.dayType === "ride")
const completedRideDaysCount = rideDays.filter((d) => d.isCompleted).length
const completedDaysCount = enrichedDays.filter((d) => d.isCompleted).length
const progressPercentage =
trip.status === "completed"
? 100
: trip.plannedTotalDistanceMeters > 0
? Math.min(
100,
Math.round(
(actualTotalDistanceMeters / trip.plannedTotalDistanceMeters) * 100
)
)
: 0
return {
...trip,
days: enrichedDays,
plannedSegments,
elevationProfile:
computedElevationProfile.length > 0
? computedElevationProfile
: trip.elevationProfile,
actualTotalDistanceMeters,
actualTotalElevationGain,
actualTotalMovingTimeSeconds,
rideDaysCount: rideDays.length,
completedRideDaysCount,
completedDaysCount,
progressPercentage,
bikeGear,
}
}

import { db } from "@/db";
import { financialAccounts, installments, purchases, transactions } from "@/db/schema";
import { and, eq, lte, or, sql } from "drizzle-orm";
export async function getDashboardSummary(userId: string, householdId: string | null, date: Date) {
const year = date.getFullYear();
const month = date.getMonth() + 1;
const currentMonthStr = `${year}-${String(month).padStart(2, "0")}`;
// Data limite do último dia do mês vigente (ex: "2026-10-31")
const lastDayOfCurrentMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();
const lastDayOfCurrentMonthStr = `${currentMonthStr}-${String(lastDayOfCurrentMonth).padStart(2, "0")}`;
const accessAccounts = householdId
? or(eq(financialAccounts.userId, userId), eq(financialAccounts.householdId, householdId))
: eq(financialAccounts.userId, userId);
const accessTransactions = householdId
? or(eq(transactions.userId, userId), eq(transactions.householdId, householdId))
: eq(transactions.userId, userId);
// 1. Saldo inicial base das contas bancárias
const [initialBalanceRes] = await db
.select({
total: sql<string>`COALESCE(SUM(CAST(${financialAccounts.initialBalance} AS DECIMAL)), 0)`,
})
.from(financialAccounts)
.where(accessAccounts);
// 2. Total de receitas pagas ATÉ O FIM DO MÊS VIGENTE
const [paidIncomeRes] = await db
.select({
total: sql<string>`COALESCE(SUM(CAST(${transactions.amount} AS DECIMAL)), 0)`,
})
.from(transactions)
.where(
and(
accessTransactions,
eq(transactions.type, "income"),
eq(transactions.paid, true),
lte(transactions.date, lastDayOfCurrentMonthStr)
)
);
// 3. Total de despesas pagas (inclui pagamentos de faturas efetuados) ATÉ O FIM DO MÊS VIGENTE
const [paidExpenseRes] = await db
.select({
total: sql<string>`COALESCE(SUM(CAST(${transactions.amount} AS DECIMAL)), 0)`,
})
.from(transactions)
.where(
and(
accessTransactions,
eq(transactions.type, "expense"),
eq(transactions.paid, true),
lte(transactions.date, lastDayOfCurrentMonthStr)
)
);
// 4. Despesas avulsas/recorrentes pendentes do mês vigente
const [pendingExpensesRes] = await db
.select({
total: sql<string>`COALESCE(SUM(CAST(${transactions.amount} AS DECIMAL)), 0)`,
})
.from(transactions)
.where(
and(
accessTransactions,
eq(transactions.type, "expense"),
eq(transactions.paid, false),
sql`TO_CHAR(CAST(${transactions.date} AS DATE), 'YYYY-MM') = ${currentMonthStr}`
)
);
// 5. Parcelas de Cartão de Crédito pendentes no mês vigente
const [pendingCardRes] = await db
.select({
total: sql<string>`COALESCE(SUM(CAST(${installments.amount} AS DECIMAL)), 0)`,
})
.from(installments)
.innerJoin(purchases, eq(installments.purchaseId, purchases.id))
.where(
and(
eq(purchases.userId, userId),
eq(installments.paid, false),
sql`TO_CHAR(CAST(${installments.dueDate} AS DATE), 'YYYY-MM') = ${currentMonthStr}`
)
);
const initialBalance = Number(initialBalanceRes?.total || 0);
const paidIncome = Number(paidIncomeRes?.total || 0);
const paidExpense = Number(paidExpenseRes?.total || 0);
const pendingExpenses = Number(pendingExpensesRes?.total || 0);
const pendingCard = Number(pendingCardRes?.total || 0);
// Saldo bancário em conta
const realBankBalance = initialBalance + paidIncome - paidExpense;
const bankBalanceCents = Math.round(realBankBalance * 100);
// Obrigações pendentes (Outras despesas + Faturas de cartão do mês)
const pendingExpensesCents = Math.round(pendingExpenses * 100);
const pendingCardObligationsCents = Math.round(pendingCard * 100);
const totalPendingObligationsCents = pendingExpensesCents + pendingCardObligationsCents;
return {
bankBalanceCents,
totalPendingObligationsCents,
pendingExpensesCents,
pendingCardObligationsCents,
expectedAvailableBalanceCents: bankBalanceCents - totalPendingObligationsCents,
};
}

import { db } from "@/db";
import { creditCardAdjustments, creditCards, purchases } from "@/db/schema";
import { and, eq, gte, lte } from "drizzle-orm";
export type InvoiceSummary = {
cartao: typeof creditCards.$inferSelect;
parcelas: any[];
ajustes: typeof creditCardAdjustments.$inferSelect[];
total: number;
pago: number;
statusFatura: "Sem gastos" | "Fechada" | "Aberta";
};
export async function getCreditCardInvoicesSummary(
userId: string,
year: number,
monthNum: number
): Promise<InvoiceSummary[]> {
const primeiroDia = new Date(year, monthNum - 1, 1);
const ultimoDia = new Date(year, monthNum, 0);
const dataInicialStr = primeiroDia.toISOString().split("T")[0];
const dataFinalStr = ultimoDia.toISOString().split("T")[0];
const cartoes = await db.query.creditCards.findMany({
where: eq(creditCards.userId, userId),
});
const cartoesComFatura = await Promise.all(
cartoes.map(async (cartao) => {
// 1. Busca as compras e filtra as parcelas
const compras = await db.query.purchases.findMany({
where: and(
eq(purchases.creditCardId, cartao.id),
eq(purchases.userId, userId)
),
with: { installments: true, category: true },
});
const parcelasDoMes = compras.flatMap((compra) =>
compra.installments
.filter((parcela) => {
const dueDate = new Date(parcela.dueDate + "T00:00:00");
return dueDate >= primeiroDia && dueDate <= ultimoDia;
})
.map((parcela) => ({
...parcela,
purchaseDescription: compra.description,
purchaseCategory: compra.category?.name || "Sem categoria",
totalInstallments: compra.installments.length,
}))
);
// 2. Busca os Ajustes (Estornos, Taxas, Resgates)
const ajustesDoMes = await db.query.creditCardAdjustments.findMany({
where: and(
eq(creditCardAdjustments.creditCardId, cartao.id),
eq(creditCardAdjustments.userId, userId),
gte(creditCardAdjustments.date, dataInicialStr),
lte(creditCardAdjustments.date, dataFinalStr)
)
});
// 3. Cálculo matemático
const totalParcelas = parcelasDoMes.reduce((sum, p) => sum + Number(p.amount), 0);
const pagoParcelas = parcelasDoMes.filter((p) => p.paid).reduce((sum, p) => sum + Number(p.amount), 0);
let totalAjustes = 0;
ajustesDoMes.forEach((ajuste) => {
const valor = Number(ajuste.amount);
if (ajuste.type === 'refund' || ajuste.type === 'cashback') {
totalAjustes -= valor;
} else if (ajuste.type === 'fee' || ajuste.type === 'rescue') {
totalAjustes += valor;
}
});
const total = totalParcelas + totalAjustes;
const pago = pagoParcelas;
const statusFatura =
(total === 0 && pago === 0 && totalAjustes === 0)
? ("Sem gastos" as const)
: (pago >= total) ? ("Fechada" as const) : ("Aberta" as const);
return {
cartao,
parcelas: parcelasDoMes.sort((a, b) => a.dueDate.localeCompare(b.dueDate)),
ajustes: ajustesDoMes.sort((a, b) => a.date.localeCompare(b.date)),
total,
pago,
statusFatura,
};
})
);
return cartoesComFatura;
}

/**
* Converte um valor financeiro (string do DB ou float) para centavos (Inteiro).
* Ex: "10.50" -> 1050 | "0.10" -> 10 | 0.20 -> 20
*/
export function toCents(value: string | number | null | undefined): number {
if (!value) return 0;
const num = typeof value === 'string' ? parseFloat(value) : value;
if (isNaN(num)) return 0;
// Math.round previne que 0.29 * 100 vire 28.99999999
return Math.round(num * 100);
}
/**
* Converte de centavos de volta para decimal padrão do sistema (Float).
* Ex: 1050 -> 10.50
*/
export function fromCents(cents: number): number {
return cents / 100;
}
/**
* Formata um valor em centavos diretamente para Real Brasileiro (BRL) na interface.
*/
export function formatCurrency(cents: number): string {
return (cents / 100).toLocaleString('pt-BR', {
style: 'currency',
currency: 'BRL',
});
}
